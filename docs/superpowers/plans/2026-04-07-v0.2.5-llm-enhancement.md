# v0.2.5 LLM Enhancement Layer — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add optional LLM-powered understanding layer — AI title, AI progress, AI summary — displayed in a redesigned two-column session tile. User-configured OpenAI-compatible API, graceful degradation to rule layer when unconfigured.

**Architecture:** Core model is `LLMUnderstandingSnapshot` (title + progress + summary as one object). `LLMEnhancer` service consumes `SessionSignals` and produces a complete snapshot via `LLMClient` (OpenAI-compatible HTTP). Snapshots are persisted in `UnderstandingStore` (separate from existing TitleStore). UI: collapsed tile gets elastic 2-3 lines with right-side best-available summary; expanded tile uses left-right split (left: rule/facts/ops from v0.2.0, right: LLM understanding panel). LLM results are front-stage; rule results are traceable insurance.

**Tech Stack:** Swift 5.9, SwiftUI, macOS 14+, Foundation URLSession (no external deps), OpenAI-compatible chat completions API.

```swift
import Foundation

/// Complete AI understanding result for one session.
/// Produced by LLMEnhancer, persisted by UnderstandingStore.
public struct LLMUnderstandingSnapshot: Codable, Sendable, Equatable {
    public let sessionID: String
    public let title: String
    public let progress: String?
    public let summary: String?
    public let modelName: String
    public let generatedAt: Date
    /// The session's lastActiveAt when this snapshot was generated.
    /// Used for stale detection: if session.lastActiveAt > basedOnLastActiveAt, snapshot is stale.
    public let basedOnLastActiveAt: Date

    public init(sessionID: String, title: String, progress: String?,
                summary: String?, modelName: String, generatedAt: Date,
                basedOnLastActiveAt: Date) {
        self.sessionID = sessionID
        self.title = title
        self.progress = progress
        self.summary = summary
        self.modelName = modelName
        self.generatedAt = generatedAt
        self.basedOnLastActiveAt = basedOnLastActiveAt
    }
}
```

Create `Sources/ClaudeSessionHub/Services/UnderstandingStore.swift`:

```swift
import Foundation
import Observation

/// Persists LLM understanding snapshots. Separate from TitleStore (rule-based).
@Observable
public final class UnderstandingStore {
    private var snapshots: [String: LLMUnderstandingSnapshot] = [:]
    private let filePath: String

    public init(directory: String = NSHomeDirectory() + "/.claude-hub") {
        self.filePath = (directory as NSString).appendingPathComponent("understanding.json")
        load()
    }

    public func snapshot(for sessionID: String) -> LLMUnderstandingSnapshot? {
        snapshots[sessionID]
    }

    public func hasEnhancement(for sessionID: String) -> Bool {
        snapshots[sessionID] != nil
    }

    /// Returns true if the snapshot exists but is older than the session's last activity.
    public func isStale(for sessionID: String, lastActiveAt: Date) -> Bool {
        guard let snapshot = snapshots[sessionID] else { return false }
        return lastActiveAt > snapshot.basedOnLastActiveAt
    }

    public func setSnapshot(_ snapshot: LLMUnderstandingSnapshot) {
        // Dedup: skip if identical
        if let existing = snapshots[snapshot.sessionID],
           existing.title == snapshot.title &&
           existing.progress == snapshot.progress &&
           existing.summary == snapshot.summary {
            return
        }
        snapshots[snapshot.sessionID] = snapshot
        save()
    }

    private func load() {
        guard let data = FileManager.default.contents(atPath: filePath),
              let stored = try? JSONDecoder().decode([String: LLMUnderstandingSnapshot].self, from: data)
        else { return }
        snapshots = stored
    }

    private func save() {
        let dir = (filePath as NSString).deletingLastPathComponent
        try? FileManager.default.createDirectory(atPath: dir, withIntermediateDirectories: true)
        guard let data = try? JSONEncoder().encode(snapshots) else { return }
        try? data.write(to: URL(fileURLWithPath: filePath))
    }
}
```

```swift
import Foundation

/// Produces a complete LLMUnderstandingSnapshot from SessionSignals.
/// Falls back to nil when LLM is unconfigured or fails.
public struct LLMEnhancer: Sendable {
    private let config: LLMConfig

    public init(config: LLMConfig) {
        self.config = config
    }

    /// Generate a complete AI understanding snapshot for a session.
    /// Returns nil if LLM is unconfigured or all calls fail.
    public func enhance(signals: SessionSignals, rawTurns: [String] = [], basedOnLastActiveAt: Date = Date()) async -> LLMUnderstandingSnapshot? {
        guard config.isConfigured else { return nil }

        let client = LLMClient(config: config)
        let input = LLMPrompts.titleInput(from: signals, rawTurns: rawTurns)

        // Generate title (required — if this fails, return nil)
        let title: String
        do {
            let response = try await client.complete(
                systemPrompt: LLMPrompts.titleSystemPrompt,
                userMessage: input,
                maxTokens: 60
            )
            let cleaned = response.trimmingCharacters(in: .whitespacesAndNewlines)
                .trimmingCharacters(in: CharacterSet(charactersIn: "\"'""''"))
            guard !cleaned.isEmpty else { return nil }
            title = String(cleaned.prefix(50))
        } catch {
            return nil
        }

        // Generate progress (optional — failure is ok)
        var progress: String? = nil
        do {
            let response = try await client.complete(
                systemPrompt: LLMPrompts.progressSystemPrompt,
                userMessage: input,
                maxTokens: 100
            )
            let cleaned = response.trimmingCharacters(in: .whitespacesAndNewlines)
            if !cleaned.isEmpty { progress = cleaned }
        } catch {}

        // Generate summary (optional — failure is ok)
        var summary: String? = nil
        do {
            let response = try await client.complete(
                systemPrompt: LLMPrompts.summarySystemPrompt,
                userMessage: input,
                maxTokens: 200
            )
            let cleaned = response.trimmingCharacters(in: .whitespacesAndNewlines)
            if !cleaned.isEmpty { summary = cleaned }
        } catch {}

        // Brief delay between API calls for rate limiting
        try? await Task.sleep(for: .milliseconds(300))

        return LLMUnderstandingSnapshot(
            sessionID: signals.sessionID,
            title: title,
            progress: progress,
            summary: summary,
            modelName: config.modelName,
            generatedAt: Date(),
            basedOnLastActiveAt: basedOnLastActiveAt
        )
    }
}
```

```swift
/// LLM-enhance a single session. User-triggered.
@MainActor
public func enhanceWithLLM(for ref: SessionRef) async throws {
    guard settings.llmConfig.isConfigured else {
        throw LLMClient.LLMError.notConfigured
    }
    guard let provider = await coordinator.provider(for: ref.providerID) as? ClaudeProvider,
          var signals = try? await provider.extractSignals(for: ref) else { return }
    signals = signalExtractor.enrich(signals)

    let session = sessions.first { $0.ref == ref }
    let lastActiveAt = session?.lastActiveAt ?? Date()

    let enhancer = LLMEnhancer(config: settings.llmConfig)
    guard let snapshot = await enhancer.enhance(signals: signals, basedOnLastActiveAt: lastActiveAt) else { return }
    understandingStore.setSnapshot(snapshot)
}
```

```swift
/// LLM-enhance visible sessions that don't yet have snapshots.
/// Scope: current visible list (filtered by project/status/search), not global.
@MainActor
public func batchEnhanceVisible() async -> Int {
    guard settings.llmConfig.isConfigured else { return 0 }

    let candidates = visibleSessions.filter { session in
        !understandingStore.hasEnhancement(for: session.ref.sessionID)
    }

    var enhanced = 0
    for session in candidates {
        do {
            try await enhanceWithLLM(for: session.ref)
            enhanced += 1
            // Rate limit between sessions
            try? await Task.sleep(for: .milliseconds(500))
        } catch {
            continue
        }
    }
    return enhanced
}
```

```swift
                HStack(spacing: 4) {

                        .font(.system(size: 9).monospaced())
                    Text("·")
                    Text(relativeTime(snapshot.generatedAt))
```

## State Machine Summary

v0.2.0 state machine unchanged. LLM enhancement is a **separate, parallel track**:

| System | Storage | Trigger | Fallback |
|---|---|---|---|
| Rule layer | TitleStore (titles.json) | Auto on scan | sessionID prefix |
| LLM layer | UnderstandingStore (understanding.json) | Manual per-session or batch | nil (not shown) |

Display priority: `AI snapshot > rule smart title > cleaned original title`

The two systems don't interfere:
- Rule layer runs on every scan (existing behavior, untouched)
- LLM layer only runs when user explicitly triggers
- Both results persist independently
- UI picks the best available for display
